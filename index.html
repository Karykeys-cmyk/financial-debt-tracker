    function formatDate(iso) {

      if (!iso) return "—";

    function daysBetween(from, to) {

      const a = new Date(from + "T00:00:00");

      const b = new Date(to + "T00:00:00");

      if (Number.isNaN(a.getTime()) || Number.isNaN(b.getTime())) return 0;

      return Math.round((b - a) / 86400000);

    }


    function calcProfit(amount, rate, issueDate, returnDate) {

      const sum = Number(amount) || 0;

      const interest = Number(rate) || 0;

      if (!interest || !issueDate || !returnDate) return 0;

      const days = daysBetween(issueDate, returnDate);

      if (days <= 0) return 0;

      return (sum * interest / 100) * (days / 365);

    }

      const extras = Object.keys(db.contractors)

        .filter((name) => name !== "Даня" && name !== "Максюшка")

        .sort((a, b) => a.localeCompare(b, "ru"));

      return ["Даня", "Максюшка", ...extras];

    }


    function currentRecords() {

      return db.contractors[state.contractor]?.records || [];

    }


    function totals(name) {

      const records = db.contractors[name]?.records || [];

      let debit = 0;

      let credit = 0;

      records.forEach((r) => {

        if (r.type === "Кредит") credit += Number(r.amount) || 0;

        else debit += Number(r.amount) || 0;

      });

      return { debit, credit, saldo: debit - credit };

    }


    function visibleRecords() {

      const cfg = contractorConfig(state.contractor);

      return currentRecords()

        .filter((r) => {

          if (!cfg.hasRental || state.tab === "debt") {

            return r.category === "Деньги в долг";

          }

          return r.category === "Аренда помещения" && r.subCategory === state.subTab;

        })

        .sort((a, b) => (b.date || "").localeCompare(a.date || "") || (b.id || "").localeCompare(a.id || ""));

    }


    function showToast(text) {

      el.textContent = text;

      el.classList.add("show");

      clearTimeout(showToast.timer);

      showToast.timer = setTimeout(() => el.classList.remove("show"), 2200);

    }

      state.view = "profile";

    function fillSelect(el, items, selected) {

      el.innerHTML = items.map((item) => {

        const value = typeof item === "string" ? item : item.value;

        const label = typeof item === "string" ? item : item.label;

        return `<option value="${escapeHtml(value)}" ${value === selected ? "selected" : ""}>${escapeHtml(label)}</option>`;

      }).join("");

    }


    function currentCategory() {

      const cfg = contractorConfig(state.contractor);

      if (!cfg.hasRental || state.tab === "debt") return "Деньги в долг";

      return "Аренда помещения";

    }


    function currentSubcategory() {

      if (currentCategory() === "Деньги в долг") return null;

      return state.subTab;

    }


    function rentalSubs() {

      const cfg = contractorConfig(state.contractor);

      const list = ["Коммуналка"];

      if (cfg.hasDtf) list.push("Аренда ДТФ");

      list.push("Счета");

    }


    function syncRecordFields() {

      const category = document.getElementById("field-category").value;

      const sub = document.getElementById("field-subcategory").value;

      const isDebt = category === "Деньги в долг";

      const isInvoice = !isDebt && sub === "Счета";

      const isPeriod = !isDebt && (sub === "Коммуналка" || sub === "Аренда ДТФ");


      document.getElementById("subcategory-field").classList.toggle("hidden", isDebt);

      document.getElementById("month-field").classList.toggle("hidden", !isPeriod);

      document.getElementById("year-field").classList.toggle("hidden", !isPeriod);

      document.getElementById("invoice-field").classList.toggle("hidden", !isInvoice);

      document.getElementById("comment-field").classList.toggle("hidden", !isInvoice);

      document.getElementById("issue-field").classList.toggle("hidden", !isDebt);

      document.getElementById("return-field").classList.toggle("hidden", !isDebt);

      document.getElementById("rate-field").classList.toggle("hidden", !isDebt);

      document.getElementById("profit-preview").classList.toggle("hidden", !isDebt);

      updateProfitPreview();

    }


    function updateProfitPreview() {

      const amount = document.getElementById("field-amount").value;

      const rate = document.getElementById("field-rate").value;

      const issue = document.getElementById("field-issue").value;

      const ret = document.getElementById("field-return").value;

      const profit = calcProfit(amount, rate, issue, ret);

    }


    function setRecordType(type) {

      state.recordType = type;

      document.getElementById("type-debit").classList.toggle("active", type === "Дебет");

      document.getElementById("type-credit").classList.toggle("active", type === "Кредит");

    function openRecordModal(id) {

      const cfg = contractorConfig(state.contractor);

      const form = document.getElementById("record-form");

      form.reset();

      document.getElementById("record-error").classList.remove("show");

      state.editingId = id || null;

      document.getElementById("record-title").textContent = id ? "Редактировать запись" : "Добавить запись";


      const cats = cfg.hasRental ? ["Аренда помещения", "Деньги в долг"] : ["Деньги в долг"];

      fillSelect(document.getElementById("field-category"), cats, currentCategory());

      fillSelect(document.getElementById("field-subcategory"), rentalSubs(), currentSubcategory() || "Коммуналка");

      fillSelect(document.getElementById("field-month"), MONTHS, MONTHS[new Date().getMonth()]);

      const years = [];

      for (let y = 2020; y <= 2030; y++) years.push(String(y));

      fillSelect(document.getElementById("field-year"), years, String(new Date().getFullYear()));


      document.getElementById("field-date").value = todayIso();

      document.getElementById("field-issue").value = todayIso();

      document.getElementById("field-rate").value = "0";

      setRecordType("Дебет");


      if (id) {

        const rec = currentRecords().find((r) => r.id === id);

        if (rec) {

          document.getElementById("field-category").value = rec.category;

          if (rec.subCategory) document.getElementById("field-subcategory").value = rec.subCategory;

          document.getElementById("field-amount").value = rec.amount;

          document.getElementById("field-date").value = rec.date || "";

          setRecordType(rec.type || "Дебет");

          if (rec.month) {

            const [monthName, year] = rec.month.split(" ");

            if (MONTHS.includes(monthName)) document.getElementById("field-month").value = monthName;

            if (year) document.getElementById("field-year").value = String(rec.year || year);

          } else if (rec.year) {

            document.getElementById("field-year").value = String(rec.year);

          }

          document.getElementById("field-invoice").value = rec.invoiceNumber || "";

          document.getElementById("field-comment").value = rec.comment || "";

          document.getElementById("field-issue").value = rec.issueDate || rec.date || "";

          document.getElementById("field-return").value = rec.returnDate || "";

          document.getElementById("field-rate").value = rec.interestRate ?? 0;

        }

      }


      document.getElementById("category-field").classList.toggle("hidden", !cfg.hasRental);

      syncRecordFields();

      document.getElementById("record-overlay").classList.add("open");

      document.getElementById("field-amount").focus();

    }


    function closeRecordModal() {

      document.getElementById("record-overlay").classList.remove("open");

      state.editingId = null;

    }


    function submitRecord(event) {

      event.preventDefault();

      const error = document.getElementById("record-error");

      const amount = Number(document.getElementById("field-amount").value);

      const date = document.getElementById("field-date").value;

      if (!document.getElementById("field-amount").value || Number.isNaN(amount) || amount <= 0) {

        error.textContent = "Укажите сумму";

        error.classList.add("show");

        return;

      }

      if (!date) {

        error.textContent = "Укажите дату";

        error.classList.add("show");

        return;

      }


      const category = document.getElementById("field-category").value;

      const subCategory = category === "Аренда помещения" ? document.getElementById("field-subcategory").value : null;

      const monthName = document.getElementById("field-month").value;

      const year = Number(document.getElementById("field-year").value);

      const isDebt = category === "Деньги в долг";

      const isInvoice = subCategory === "Счета";

      const isPeriod = subCategory === "Коммуналка" || subCategory === "Аренда ДТФ";

      const issueDate = document.getElementById("field-issue").value || null;

      const returnDate = document.getElementById("field-return").value || null;

      const interestRate = Number(document.getElementById("field-rate").value) || 0;


      const record = {

        id: state.editingId || uid(),

        category,

        subCategory,

        type: state.recordType,

        amount,

        date,

        month: isPeriod ? `${monthName} ${year}` : null,

        year: isPeriod ? year : null,

        invoiceNumber: isInvoice ? document.getElementById("field-invoice").value.trim() : null,

        comment: isInvoice ? document.getElementById("field-comment").value.trim() : "",

        issueDate: isDebt ? issueDate : null,

        returnDate: isDebt ? returnDate : null,

        interestRate: isDebt ? interestRate : null,

        profit: isDebt ? calcProfit(amount, interestRate, issueDate, returnDate) : 0

      };


      const list = db.contractors[state.contractor].records;

      const idx = list.findIndex((r) => r.id === record.id);

      if (idx >= 0) list[idx] = record;

      else list.push(record);

      saveData(db);

      closeRecordModal();

      render();

      showToast(idx >= 0 ? "Запись обновлена" : "Запись сохранена");

    }


    function askDelete(id) {

      state.pendingDeleteId = id;

      document.getElementById("confirm-overlay").classList.add("open");

    }


    function confirmDelete() {

      const id = state.pendingDeleteId;

      if (!id) return;

      db.contractors[state.contractor].records = currentRecords().filter((r) => r.id !== id);

      saveData(db);

      state.pendingDeleteId = null;

      document.getElementById("confirm-overlay").classList.remove("open");

      render();

      showToast("Запись удалена");

    }

    function submitContractor(event) {

      event.preventDefault();

      const error = document.getElementById("contractor-error");

      const name = document.getElementById("field-contractor-name").value.trim();

      if (!name) {

        error.textContent = "Введите имя контрагента";

        error.classList.add("show");

        return;

      }

      if (db.contractors[name]) {

        error.textContent = "Контрагент с таким именем уже существует";

        error.classList.add("show");

        return;

      }

      db.contractors[name] = { records: [] };

      saveData(db);

      closeContractorModal();

      render();

      showToast("Контрагент добавлен");

    }
